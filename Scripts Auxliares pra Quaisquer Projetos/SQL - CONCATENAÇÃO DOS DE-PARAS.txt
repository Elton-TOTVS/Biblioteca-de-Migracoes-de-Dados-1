/*==================================================================================================
    CONCATENAÇÃO DOS DE-PARAS
    DESTINO: ZMIGRA_GENNESIS_PRD2
==================================================================================================*/

SET NOCOUNT ON;

DECLARE @BancoDestino SYSNAME = 'ZMIGRA_GENNESIS_PRD2';
DECLARE @Schema       SYSNAME = 'dbo';


/*==================================================================================================
    BANCOS DE ORIGEM
==================================================================================================*/
DECLARE @Bancos TABLE
(
    Id      INT IDENTITY(1,1),
    Banco   SYSNAME
);

INSERT INTO @Bancos (Banco)
VALUES
    ('ZMIGRA_GENNESIS_ABRASIL_PRD'),
    ('ZMIGRA_GENNESIS_AVALIACOES_PRD'),
    ('ZMIGRA_GENNESIS_CONSULT_PRD'),
    ('ZMIGRA_GENNESIS_ECONTECX_PRD'),
    ('ZMIGRA_GENNESIS_ENGENH_PRD'),
    ('ZMIGRA_GENNESIS_ENGTHA_PRD'),
    ('ZMIGRA_GENNESIS_METRICA_PRD'),
    ('ZMIGRA_GENNESIS_T2_PRD'),
    ('ZMIGRA_GENNESIS_TESLA_PRD');


/*==================================================================================================
    TABELAS A SEREM CONCATENADAS
==================================================================================================*/
DECLARE @Tabelas TABLE
(
    Id      INT IDENTITY(1,1),
    Tabela  SYSNAME
);

INSERT INTO @Tabelas (Tabela)
VALUES
    ('ZDEPARA_BANCOS'),
    ('ZDEPARA_COLIGADAS'),
    ('ZDEPARA_EVENTOS'),
    ('ZDEPARA_FUNCOES'),
    ('ZDEPARA_HORARIO'),
    ('ZDEPARA_PERIODO_FOLHA'),
    ('ZDEPARA_SECOES'),
    ('ZDEPARA_SINDICATOS'),
    ('ZDEPARA_SITUACAO');


/*==================================================================================================
    VARIÁVEIS
==================================================================================================*/
DECLARE
    @Tabela         SYSNAME,
    @Banco          SYSNAME,
    @PrimeiroBanco  SYSNAME,
    @SQL            NVARCHAR(MAX),
    @Colunas        NVARCHAR(MAX),
    @TemIdentity    BIT;


/* Primeiro banco será usado como estrutura base */
SELECT TOP 1
    @PrimeiroBanco = Banco
FROM @Bancos
ORDER BY Id;


/*==================================================================================================
    PROCESSAMENTO DAS TABELAS
==================================================================================================*/
DECLARE CursorTabelas CURSOR LOCAL FAST_FORWARD
FOR
    SELECT Tabela
    FROM @Tabelas
    ORDER BY Id;

OPEN CursorTabelas;

FETCH NEXT FROM CursorTabelas
INTO @Tabela;

WHILE @@FETCH_STATUS = 0
BEGIN

    PRINT '===============================================================================';
    PRINT 'PROCESSANDO: ' + @Tabela;
    PRINT '===============================================================================';


    /*==============================================================================================
        1. REMOVE SOMENTE A TABELA CORRESPONDENTE DO BANCO DESTINO
    ==============================================================================================*/
    SET @SQL = N'
        USE ' + QUOTENAME(@BancoDestino) + N';

        IF OBJECT_ID(
            N''' + QUOTENAME(@Schema) + N'.' + QUOTENAME(@Tabela) + N''',
            N''U''
        ) IS NOT NULL
        BEGIN
            DROP TABLE '
            + QUOTENAME(@Schema) + N'.' + QUOTENAME(@Tabela) + N';
        END;
    ';

    EXEC sys.sp_executesql @SQL;


    /*==============================================================================================
        2. CRIA A TABELA NO DESTINO E COPIA O PRIMEIRO BANCO
    ==============================================================================================*/
    SET @SQL = N'
        SELECT *
        INTO '
        + QUOTENAME(@BancoDestino)
        + N'.'
        + QUOTENAME(@Schema)
        + N'.'
        + QUOTENAME(@Tabela)
        + N'
        FROM '
        + QUOTENAME(@PrimeiroBanco)
        + N'.'
        + QUOTENAME(@Schema)
        + N'.'
        + QUOTENAME(@Tabela)
        + N';
    ';

    EXEC sys.sp_executesql @SQL;

    PRINT 'Copiado: ' + @PrimeiroBanco;


    /*==============================================================================================
        3. DESCOBRE AS COLUNAS DA TABELA CRIADA
    ==============================================================================================*/
    SET @Colunas = NULL;
    SET @TemIdentity = 0;

    SET @SQL = N'
        SELECT
            @Colunas = STRING_AGG(
                CAST(QUOTENAME(C.name) AS NVARCHAR(MAX)),
                N'', ''
            ) WITHIN GROUP (
                ORDER BY C.column_id
            ),

            @TemIdentity =
                CONVERT(
                    BIT,
                    MAX(
                        CASE
                            WHEN C.is_identity = 1 THEN 1
                            ELSE 0
                        END
                    )
                )

        FROM '
        + QUOTENAME(@BancoDestino) + N'.sys.columns C

        INNER JOIN '
        + QUOTENAME(@BancoDestino) + N'.sys.tables T
            ON T.object_id = C.object_id

        INNER JOIN '
        + QUOTENAME(@BancoDestino) + N'.sys.schemas S
            ON S.schema_id = T.schema_id

        WHERE
            T.name = @Tabela
            AND S.name = @Schema;
    ';

    EXEC sys.sp_executesql
        @SQL,
        N'
            @Tabela SYSNAME,
            @Schema SYSNAME,
            @Colunas NVARCHAR(MAX) OUTPUT,
            @TemIdentity BIT OUTPUT
        ',
        @Tabela = @Tabela,
        @Schema = @Schema,
        @Colunas = @Colunas OUTPUT,
        @TemIdentity = @TemIdentity OUTPUT;


    /*==============================================================================================
        4. COPIA OS DEMAIS BANCOS
    ==============================================================================================*/
    DECLARE CursorBancos CURSOR LOCAL FAST_FORWARD
    FOR
        SELECT Banco
        FROM @Bancos
        WHERE Id > 1
        ORDER BY Id;

    OPEN CursorBancos;

    FETCH NEXT FROM CursorBancos
    INTO @Banco;

    WHILE @@FETCH_STATUS = 0
    BEGIN

        SET @SQL = N'';


        /* Caso a tabela tenha IDENTITY */
        IF @TemIdentity = 1
        BEGIN
            SET @SQL += N'
                SET IDENTITY_INSERT '
                + QUOTENAME(@BancoDestino)
                + N'.'
                + QUOTENAME(@Schema)
                + N'.'
                + QUOTENAME(@Tabela)
                + N' ON;
            ';
        END;


        SET @SQL += N'

            INSERT INTO '
            + QUOTENAME(@BancoDestino)
            + N'.'
            + QUOTENAME(@Schema)
            + N'.'
            + QUOTENAME(@Tabela)
            + N'
            (
                ' + @Colunas + N'
            )

            SELECT
                ' + @Colunas + N'
            FROM '
            + QUOTENAME(@Banco)
            + N'.'
            + QUOTENAME(@Schema)
            + N'.'
            + QUOTENAME(@Tabela)
            + N';
        ';


        IF @TemIdentity = 1
        BEGIN
            SET @SQL += N'
                SET IDENTITY_INSERT '
                + QUOTENAME(@BancoDestino)
                + N'.'
                + QUOTENAME(@Schema)
                + N'.'
                + QUOTENAME(@Tabela)
                + N' OFF;
            ';
        END;


        EXEC sys.sp_executesql @SQL;

        PRINT 'Copiado: ' + @Banco;


        FETCH NEXT FROM CursorBancos
        INTO @Banco;
    END;


    CLOSE CursorBancos;
    DEALLOCATE CursorBancos;


    PRINT 'FINALIZADO: ' + @Tabela;
    PRINT '';


    FETCH NEXT FROM CursorTabelas
    INTO @Tabela;
END;


CLOSE CursorTabelas;
DEALLOCATE CursorTabelas;


/*==================================================================================================
    VALIDAÇÃO FINAL
==================================================================================================*/

FROM ZMIGRA_GENNESIS_PRD2.dbo.ZDEPARA_BANCOS
