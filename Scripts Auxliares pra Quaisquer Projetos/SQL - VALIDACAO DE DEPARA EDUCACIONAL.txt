/*
================================================================================
VALIDADOR DE-PARA - SESC_EDUSOFT_MS - VERSÃO 3.0
================================================================================
Objetivo:
  Revisão 3.0: correção das agregações da cobertura e auditoria bidirecional dos campos PARA.
  1. Localizar campos NULL ou vazios em todas as tabelas dbo.ZDEPARA_*.
  2. Localizar códigos de destino sem correspondência em outros de-paras.
  3. Localizar combinações incompatíveis entre curso, habilitação, grade,
     habilitação por filial, disciplina, turno, turma, prédio, bloco e sala.
  4. Comparar bidirecionalmente os campos PARA entre SHABILITACAOFILIAL e
     TURMAS, incluindo matrizes aplicadas sem turmas na filial.
  5. Localizar descrições divergentes nos relacionamentos em que a estrutura
     fornecida permite identificar o texto de referência.
  6. Retornar:
       - um resumo geral;
       - uma consulta consolidada;
       - uma consulta/result set separado para cada tabela com pendências.

Características:
  - Somente leitura nas tabelas permanentes.
  - Cria apenas tabelas temporárias.
  - Pode ser executado como um único arquivo .sql no VS Code.
================================================================================
*/

USE [SESC_EDUSOFT_MS];
GO

SET NOCOUNT ON;
SET XACT_ABORT ON;

DECLARE
    @VALIDAR_TODOS_OS_CAMPOS_NULOS BIT = 1,
    @GERAR_RESULTADO_POR_TABELA    BIT = 1;

DROP TABLE IF EXISTS #PENDENCIAS;
CREATE TABLE #PENDENCIAS
(
    ID                  BIGINT IDENTITY(1,1) NOT NULL,
    TABELA_DEPARA       SYSNAME              NOT NULL,
    REGRA               NVARCHAR(200)        NOT NULL,
    TIPO_PENDENCIA      VARCHAR(50)          NOT NULL,
    SEVERIDADE          VARCHAR(10)          NOT NULL,
    CHAVE_REGISTRO      NVARCHAR(2000)       NULL,
    CAMPO               SYSNAME              NULL,
    VALOR_INFORMADO     NVARCHAR(4000)       NULL,
    VALOR_REFERENCIA    NVARCHAR(4000)       NULL,
    TABELA_REFERENCIA   SYSNAME              NULL,
    DETALHE             NVARCHAR(4000)       NULL
);

DROP TABLE IF EXISTS #TABELAS_DEPARA;
CREATE TABLE #TABELAS_DEPARA
(
    TABELA       SYSNAME       NOT NULL PRIMARY KEY,
    CHAVE_EXPR   NVARCHAR(MAX) NOT NULL
);

DROP TABLE IF EXISTS #RESUMO_TABELAS;
CREATE TABLE #RESUMO_TABELAS
(
    TABELA             SYSNAME NOT NULL PRIMARY KEY,
    TOTAL_REGISTROS    BIGINT  NOT NULL
);

/* ============================================================================
   1. DESCOBERTA DAS TABELAS E CONSTRUÇÃO DE UMA CHAVE VISUAL
   Usa os quatro primeiros campos que não terminam em "_PARA".
============================================================================ */
DECLARE
    @TABELA_ATUAL SYSNAME,
    @CHAVE_EXPR   NVARCHAR(MAX),
    @SQL          NVARCHAR(MAX);

DECLARE CUR_TABELAS CURSOR LOCAL FAST_FORWARD FOR
    SELECT T.NAME
    FROM SYS.TABLES T
    INNER JOIN SYS.SCHEMAS S
        ON S.SCHEMA_ID = T.SCHEMA_ID
    WHERE S.NAME = N'dbo'
      AND T.NAME LIKE N'ZDEPARA[_]%'
    ORDER BY T.NAME;

OPEN CUR_TABELAS;
FETCH NEXT FROM CUR_TABELAS INTO @TABELA_ATUAL;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @CHAVE_EXPR = NULL;

    SELECT
        @CHAVE_EXPR =
            STRING_AGG(CONVERT(NVARCHAR(MAX), X.PARTE), N' + N''; '' + ')
            WITHIN GROUP (ORDER BY X.COLUMN_ID)
    FROM
    (
        SELECT TOP (4)
            C.COLUMN_ID,
            N'N''' + REPLACE(C.NAME, N'''', N'''''') + N'='' + ' +
            N'COALESCE(TRY_CONVERT(NVARCHAR(300), D.' + QUOTENAME(C.NAME) +
            N'), N''<NULL>'')' AS PARTE
        FROM SYS.COLUMNS C
        INNER JOIN SYS.TABLES T
            ON T.OBJECT_ID = C.OBJECT_ID
        INNER JOIN SYS.SCHEMAS S
            ON S.SCHEMA_ID = T.SCHEMA_ID
        WHERE S.NAME = N'dbo'
          AND T.NAME = @TABELA_ATUAL
          AND C.IS_COMPUTED = 0
          AND C.NAME NOT LIKE N'%[_]PARA'
        ORDER BY C.COLUMN_ID
    ) X;

    IF @CHAVE_EXPR IS NULL
        SET @CHAVE_EXPR = N'N''Registro sem chave visual identificável''';

    INSERT INTO #TABELAS_DEPARA (TABELA, CHAVE_EXPR)
    VALUES (@TABELA_ATUAL, @CHAVE_EXPR);

    SET @SQL = N'
        INSERT INTO #RESUMO_TABELAS (TABELA, TOTAL_REGISTROS)
        SELECT @P_TABELA, COUNT_BIG(*)
        FROM dbo.' + QUOTENAME(@TABELA_ATUAL) + N';';

    EXEC SYS.SP_EXECUTESQL
        @SQL,
        N'@P_TABELA SYSNAME',
        @P_TABELA = @TABELA_ATUAL;

    FETCH NEXT FROM CUR_TABELAS INTO @TABELA_ATUAL;
END;

CLOSE CUR_TABELAS;
DEALLOCATE CUR_TABELAS;

/* ============================================================================
   2. CAMPOS NULL OU VAZIOS
   - ERRO: campos de destino/código que aparentam ser obrigatórios.
   - AVISO: demais campos, quando @VALIDAR_TODOS_OS_CAMPOS_NULOS = 1.
============================================================================ */
DECLARE
    @CAMPO_ATUAL       SYSNAME,
    @SEVERIDADE_ATUAL VARCHAR(10);

DECLARE CUR_CAMPOS CURSOR LOCAL FAST_FORWARD FOR
    SELECT
        TD.TABELA,
        C.NAME AS CAMPO,
        TD.CHAVE_EXPR,
        CASE
            WHEN
                (
                    C.NAME LIKE N'%[_]PARA'
                    AND C.NAME NOT LIKE N'%NOME%'
                    AND C.NAME NOT LIKE N'%DESCRICAO%'
                )
                OR C.NAME IN
                (
                    N'FILIAL_PARA',
                    N'PREDIO_PARA',
                    N'BLOCO_PARA',
                    N'IDHABILITACAOFILIAL_PARA'
                )
            THEN 'ERRO'
            ELSE 'AVISO'
        END AS SEVERIDADE
    FROM #TABELAS_DEPARA TD
    INNER JOIN SYS.TABLES T
        ON T.NAME = TD.TABELA
    INNER JOIN SYS.SCHEMAS S
        ON S.SCHEMA_ID = T.SCHEMA_ID
       AND S.NAME = N'dbo'
    INNER JOIN SYS.COLUMNS C
        ON C.OBJECT_ID = T.OBJECT_ID
    WHERE C.IS_COMPUTED = 0
      AND
      (
          @VALIDAR_TODOS_OS_CAMPOS_NULOS = 1
          OR
          (
              (
                  C.NAME LIKE N'%[_]PARA'
                  AND C.NAME NOT LIKE N'%NOME%'
                  AND C.NAME NOT LIKE N'%DESCRICAO%'
              )
              OR C.NAME IN
              (
                  N'FILIAL_PARA',
                  N'PREDIO_PARA',
                  N'BLOCO_PARA',
                  N'IDHABILITACAOFILIAL_PARA'
              )
          )
      )
    ORDER BY TD.TABELA, C.COLUMN_ID;

OPEN CUR_CAMPOS;
FETCH NEXT FROM CUR_CAMPOS
    INTO @TABELA_ATUAL, @CAMPO_ATUAL, @CHAVE_EXPR, @SEVERIDADE_ATUAL;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @SQL = N'
        INSERT INTO #PENDENCIAS
        (
            TABELA_DEPARA,
            REGRA,
            TIPO_PENDENCIA,
            SEVERIDADE,
            CHAVE_REGISTRO,
            CAMPO,
            VALOR_INFORMADO,
            VALOR_REFERENCIA,
            TABELA_REFERENCIA,
            DETALHE
        )
        SELECT
            @P_TABELA,
            N''Preenchimento obrigatório'',
            N''CAMPO_NULO_OU_VAZIO'',
            @P_SEVERIDADE,
            ' + @CHAVE_EXPR + N',
            @P_CAMPO,
            TRY_CONVERT(NVARCHAR(4000), D.' + QUOTENAME(@CAMPO_ATUAL) + N'),
            NULL,
            NULL,
            CASE
                WHEN @P_SEVERIDADE = ''ERRO''
                    THEN N''Campo de destino/código sem preenchimento.''
                ELSE N''Campo sem preenchimento. Verifique se o NULL é permitido pela regra de negócio.''
            END
        FROM dbo.' + QUOTENAME(@TABELA_ATUAL) + N' D
        WHERE D.' + QUOTENAME(@CAMPO_ATUAL) + N' IS NULL
           OR NULLIF
              (
                  LTRIM(RTRIM(TRY_CONVERT(NVARCHAR(4000), D.' +
                  QUOTENAME(@CAMPO_ATUAL) + N'))),
                  N''''
              ) IS NULL;';

    EXEC SYS.SP_EXECUTESQL
        @SQL,
        N'@P_TABELA SYSNAME, @P_CAMPO SYSNAME, @P_SEVERIDADE VARCHAR(10)',
        @P_TABELA = @TABELA_ATUAL,
        @P_CAMPO = @CAMPO_ATUAL,
        @P_SEVERIDADE = @SEVERIDADE_ATUAL;

    FETCH NEXT FROM CUR_CAMPOS
        INTO @TABELA_ATUAL, @CAMPO_ATUAL, @CHAVE_EXPR, @SEVERIDADE_ATUAL;
END;

CLOSE CUR_CAMPOS;
DEALLOCATE CUR_CAMPOS;

/* 3.1 BLOCO -> PRÉDIO */
INSERT INTO #PENDENCIAS
(
    TABELA_DEPARA, REGRA, TIPO_PENDENCIA, SEVERIDADE, CHAVE_REGISTRO,
    CAMPO, VALOR_INFORMADO, VALOR_REFERENCIA, TABELA_REFERENCIA, DETALHE
)
SELECT
    N'ZDEPARA_BLOCO',
    N'Bloco deve referenciar um prédio mapeado',
    N'REFERENCIA_INEXISTENTE',
    N'ERRO',
    CONCAT
    (
        N'CODIGO=', COALESCE(CONVERT(NVARCHAR(200), B.CODIGO), N'<NULL>'),
        N'; UNIDADE=', COALESCE(CONVERT(NVARCHAR(200), B.UNIDADE), N'<NULL>')
    ),
    N'CODPREDIO_PARA',
    CONVERT(NVARCHAR(4000), B.CODPREDIO_PARA),
    NULL,
    N'ZDEPARA_SPREDIO',
    N'O código de prédio informado no bloco não foi localizado no de-para de prédios para a mesma coligada e filial.'
FROM dbo.ZDEPARA_BLOCO B
WHERE NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), B.CODPREDIO_PARA))), N'') IS NOT NULL
  AND NOT EXISTS
  (
      SELECT 1
      FROM dbo.ZDEPARA_SPREDIO P
      WHERE NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), P.CODIGO_PARA))), N'')
            = NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), B.CODPREDIO_PARA))), N'')
        AND NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), P.CODCOLIGADA))), N'')
            = NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), B.CODCOLIGADA))), N'')
        AND NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), P.CODFILIAL_PARA))), N'')
            = NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), B.CODFILIAL_PARA))), N'')
  );

      <> ISNULL(NULLIF(LTRIM(RTRIM(R.DESCRICAO_ESPERADA)), N''), N'<NULL>');

/* ============================================================================
   6. RESULT SET 1 - RESUMO POR TABELA
============================================================================ */
SELECT
    R.TABELA,
    R.TOTAL_REGISTROS,
    COUNT(P.ID) AS TOTAL_PENDENCIAS,
    SUM(CASE WHEN P.SEVERIDADE = 'ERRO'  THEN 1 ELSE 0 END) AS TOTAL_ERROS,
    SUM(CASE WHEN P.SEVERIDADE = 'AVISO' THEN 1 ELSE 0 END) AS TOTAL_AVISOS,
    CASE
        WHEN SUM(CASE WHEN P.SEVERIDADE = 'ERRO' THEN 1 ELSE 0 END) > 0
            THEN N'COM ERROS'
        WHEN COUNT(P.ID) > 0
            THEN N'SOMENTE AVISOS'
        ELSE N'OK'
    END AS STATUS_VALIDACAO
FROM #RESUMO_TABELAS R
LEFT JOIN #PENDENCIAS P
    ON P.TABELA_DEPARA = R.TABELA
GROUP BY R.TABELA, R.TOTAL_REGISTROS
ORDER BY
    CASE
        WHEN SUM(CASE WHEN P.SEVERIDADE = 'ERRO' THEN 1 ELSE 0 END) > 0 THEN 1
        WHEN COUNT(P.ID) > 0 THEN 2
        ELSE 3
    END,
    R.TABELA;

/* ============================================================================
   7. RESULT SET 2 - TODAS AS PENDÊNCIAS CONSOLIDADAS
============================================================================ */
SELECT
    ID,
    TABELA_DEPARA,
    SEVERIDADE,
    TIPO_PENDENCIA,
    REGRA,
    CHAVE_REGISTRO,
    CAMPO,
    VALOR_INFORMADO,
    VALOR_REFERENCIA,
    TABELA_REFERENCIA,
    DETALHE
FROM #PENDENCIAS
ORDER BY
    CASE SEVERIDADE WHEN 'ERRO' THEN 1 ELSE 2 END,
    TABELA_DEPARA,
    TIPO_PENDENCIA,
    ID;

/* ============================================================================
   8. RESULT SET 3 - COBERTURA SHABILITACAOFILIAL x TURMAS
   Mostra cada matriz aplicada e quantas turmas existem:
     - na filial;
     - utilizando o mesmo ID;
     - utilizando todos os campos PARA exatamente iguais.

   Implementação:
     As contagens são pré-agregadas exclusivamente com colunas de ZDEPARA_TURMAS
     e depois relacionadas à SHABILITACAOFILIAL. Isso evita o erro SQL Server:
     "Multiple columns are specified in an aggregated expression containing
      an outer reference".
============================================================================ */
;WITH HF AS
(
    SELECT
        H.*,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODCOLIGADA))), N'')          AS K_COLIGADA,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODFILIAL_PARA))), N'')      AS K_FILIAL,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.NIVELENSINO))), N'')         AS K_NIVEL,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODCURSO_PARA))), N'')       AS K_CURSO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODHABILITACAO_PARA))), N'') AS K_HABILITACAO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODGRADE_PARA))), N'')       AS K_GRADE,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODTURNO_PARA))), N'')       AS K_TURNO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), H.CODIGO_PARA))), N'')         AS K_ID_MATRIZ
    FROM dbo.ZDEPARA_SHABILITACAOFILIAL H
),
TURMAS AS
(
    SELECT
        T.*,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.CODCOLIGADA))), N'')              AS K_COLIGADA,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.FILIAL_PARA))), N'')              AS K_FILIAL,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.NIVEL_ENSINO))), N'')             AS K_NIVEL,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.CODCURSO_PARA))), N'')            AS K_CURSO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.CODHABILITACAO_PARA))), N'')      AS K_HABILITACAO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.CODGRADE_PARA))), N'')            AS K_GRADE,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.CODTURNO_PARA))), N'')            AS K_TURNO,
        NULLIF(LTRIM(RTRIM(CONVERT(NVARCHAR(200), T.IDHABILITACAOFILIAL_PARA))), N'') AS K_ID_MATRIZ
    FROM dbo.ZDEPARA_TURMAS T
),
TURMAS_POR_FILIAL AS
(
    SELECT
        ISNULL(K_COLIGADA, N'<NULL>') AS K_COLIGADA,
        ISNULL(K_FILIAL, N'<NULL>')   AS K_FILIAL,
        COUNT_BIG(*)                  AS QTD_TURMAS_FILIAL
    FROM TURMAS
    GROUP BY
        ISNULL(K_COLIGADA, N'<NULL>'),
        ISNULL(K_FILIAL, N'<NULL>')
),
TURMAS_POR_ID AS
(
    SELECT
        ISNULL(K_COLIGADA, N'<NULL>') AS K_COLIGADA,
        ISNULL(K_ID_MATRIZ, N'<NULL>') AS K_ID_MATRIZ,
        COUNT_BIG(*)                   AS QTD_TURMAS_USANDO_ID
    FROM TURMAS
    GROUP BY
        ISNULL(K_COLIGADA, N'<NULL>'),
        ISNULL(K_ID_MATRIZ, N'<NULL>')
),
TURMAS_POR_COMBINACAO_EXATA AS
(
    SELECT
        ISNULL(K_COLIGADA, N'<NULL>')    AS K_COLIGADA,
        ISNULL(K_FILIAL, N'<NULL>')      AS K_FILIAL,
        ISNULL(K_NIVEL, N'<NULL>')       AS K_NIVEL,
        ISNULL(K_CURSO, N'<NULL>')       AS K_CURSO,
        ISNULL(K_HABILITACAO, N'<NULL>') AS K_HABILITACAO,
        ISNULL(K_GRADE, N'<NULL>')       AS K_GRADE,
        ISNULL(K_TURNO, N'<NULL>')       AS K_TURNO,
        ISNULL(K_ID_MATRIZ, N'<NULL>')   AS K_ID_MATRIZ,
        COUNT_BIG(*)                     AS QTD_TURMAS_COMBINACAO_EXATA
    FROM TURMAS
    GROUP BY
        ISNULL(K_COLIGADA, N'<NULL>'),
        ISNULL(K_FILIAL, N'<NULL>'),
        ISNULL(K_NIVEL, N'<NULL>'),
        ISNULL(K_CURSO, N'<NULL>'),
        ISNULL(K_HABILITACAO, N'<NULL>'),
        ISNULL(K_GRADE, N'<NULL>'),
        ISNULL(K_TURNO, N'<NULL>'),
        ISNULL(K_ID_MATRIZ, N'<NULL>')
)
SELECT
    HF.CODIGO              AS CODIGO_DE,
    HF.CODIGO_PARA         AS ID_MATRIZ_APLICADA_PARA,
    HF.CODCOLIGADA,
    HF.CODFILIAL_PARA      AS FILIAL_PARA,
    HF.NIVELENSINO         AS NIVEL_ENSINO_PARA,
    HF.CODCURSO_PARA,
    HF.CODHABILITACAO_PARA,
    HF.CODGRADE_PARA,
    HF.CODTURNO_PARA,
    ISNULL(F.QTD_TURMAS_FILIAL, 0)          AS QTD_TURMAS_FILIAL,
    ISNULL(I.QTD_TURMAS_USANDO_ID, 0)       AS QTD_TURMAS_USANDO_ID,
    ISNULL(E.QTD_TURMAS_COMBINACAO_EXATA, 0) AS QTD_TURMAS_COMBINACAO_EXATA,
    CASE
        WHEN ISNULL(F.QTD_TURMAS_FILIAL, 0) = 0
            THEN N'ERRO - FILIAL SEM TURMAS'
        WHEN ISNULL(I.QTD_TURMAS_USANDO_ID, 0) = 0
            THEN N'ERRO - MATRIZ APLICADA NÃO UTILIZADA'
        WHEN ISNULL(E.QTD_TURMAS_COMBINACAO_EXATA, 0) = 0
            THEN N'ERRO - ID UTILIZADO COM CAMPOS PARA DIVERGENTES'
        WHEN ISNULL(E.QTD_TURMAS_COMBINACAO_EXATA, 0)
             < ISNULL(I.QTD_TURMAS_USANDO_ID, 0)
            THEN N'ERRO - EXISTEM TURMAS DIVERGENTES PARA O MESMO ID'
        ELSE N'OK'
    END AS STATUS_COBERTURA
FROM HF
LEFT JOIN TURMAS_POR_FILIAL F
    ON F.K_COLIGADA = ISNULL(HF.K_COLIGADA, N'<NULL>')
   AND F.K_FILIAL   = ISNULL(HF.K_FILIAL, N'<NULL>')
LEFT JOIN TURMAS_POR_ID I
    ON I.K_COLIGADA = ISNULL(HF.K_COLIGADA, N'<NULL>')
   AND I.K_ID_MATRIZ = ISNULL(HF.K_ID_MATRIZ, N'<NULL>')
LEFT JOIN TURMAS_POR_COMBINACAO_EXATA E
    ON E.K_COLIGADA    = ISNULL(HF.K_COLIGADA, N'<NULL>')
   AND E.K_FILIAL      = ISNULL(HF.K_FILIAL, N'<NULL>')
   AND E.K_NIVEL       = ISNULL(HF.K_NIVEL, N'<NULL>')
   AND E.K_CURSO       = ISNULL(HF.K_CURSO, N'<NULL>')
   AND E.K_HABILITACAO = ISNULL(HF.K_HABILITACAO, N'<NULL>')
   AND E.K_GRADE       = ISNULL(HF.K_GRADE, N'<NULL>')
   AND E.K_TURNO       = ISNULL(HF.K_TURNO, N'<NULL>')
   AND E.K_ID_MATRIZ   = ISNULL(HF.K_ID_MATRIZ, N'<NULL>')
ORDER BY
    CASE
        WHEN ISNULL(F.QTD_TURMAS_FILIAL, 0) = 0 THEN 1
        WHEN ISNULL(I.QTD_TURMAS_USANDO_ID, 0) = 0 THEN 2
        WHEN ISNULL(E.QTD_TURMAS_COMBINACAO_EXATA, 0)
             < ISNULL(I.QTD_TURMAS_USANDO_ID, 0) THEN 3
        ELSE 4
    END,
    HF.CODCOLIGADA,
    HF.CODFILIAL_PARA,
    HF.CODIGO_PARA;


/* ============================================================================
   9. RESULT SETS ADICIONAIS - UMA CONSULTA PARA CADA TABELA COM PENDÊNCIA
============================================================================ */
IF @GERAR_RESULTADO_POR_TABELA = 1
BEGIN
    DECLARE @TABELA_RESULTADO SYSNAME;

    DECLARE CUR_RESULTADOS CURSOR LOCAL FAST_FORWARD FOR
        SELECT DISTINCT TABELA_DEPARA
        FROM #PENDENCIAS
        ORDER BY TABELA_DEPARA;

    OPEN CUR_RESULTADOS;
    FETCH NEXT FROM CUR_RESULTADOS INTO @TABELA_RESULTADO;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        SELECT
            CONCAT(N'PENDÊNCIAS: ', @TABELA_RESULTADO) AS RESULTADO_ATUAL;

        SELECT
            ID,
            TABELA_DEPARA,
            SEVERIDADE,
            TIPO_PENDENCIA,
            REGRA,
            CHAVE_REGISTRO,
            CAMPO,
            VALOR_INFORMADO,
            VALOR_REFERENCIA,
            TABELA_REFERENCIA,
            DETALHE
        FROM #PENDENCIAS
        WHERE TABELA_DEPARA = @TABELA_RESULTADO
        ORDER BY
            CASE SEVERIDADE WHEN 'ERRO' THEN 1 ELSE 2 END,
            TIPO_PENDENCIA,
            ID;

        FETCH NEXT FROM CUR_RESULTADOS INTO @TABELA_RESULTADO;
    END;

    CLOSE CUR_RESULTADOS;
    DEALLOCATE CUR_RESULTADOS;
END;
